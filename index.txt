function renderDashboard(){
  const k = state.knowledge;
  const total = k.length;
  const active = k.filter(x=>x.status==='Active').length;
  const review = k.filter(x=>x.status==='UnderReview').length;
  const openGaps = state.failed.filter(f=>f.status==='مفتوح').length;

  const kpis = [
    [total, 'إجمالي المعارف'],
    [active, 'معارف نشطة'],
    [review, 'بانتظار المراجعة'],
    [openGaps, 'فجوات مفتوحة (أسئلة فاشلة)'],
  ];
  document.getElementById('kpiGrid').innerHTML = kpis.map(([n,l])=>
    `<div class="card kpi"><div class="num">${n}</div><div class="lbl">${esc(l)}</div></div>`
  ).join('');

  const activity = [];
  k.forEach(x=> activity.push({t:x.updatedAt||x.createdAt, txt:`تحديث معرفة: «${x.title}»`, badge:STATUS_BADGE[x.status], badgeTxt:STATUS_LABELS[x.status]}));
  state.failed.forEach(x=> activity.push({t:x.createdAt, txt:`سؤال فاشل: «${x.questionText}»`, badge:'b-draft', badgeTxt:x.status}));
  state.corrections.forEach(x=> activity.push({t:x.createdAt, txt:`تصحيح موظف: «${x.questionText}»`, badge:'b-review', badgeTxt:x.reasonCode}));
  activity.sort((a,b)=> (b.t||'').localeCompare(a.t||''));
  const recentEl = document.getElementById('recentActivity');
  recentEl.innerHTML = activity.slice(0,7).map(a=>
    `<div class="flex-between" style="border-bottom:1px solid var(--line); padding-bottom:8px;">
       <div style="font-size:12.5px;">${esc(a.txt)}</div>
       <span class="badge ${a.badge}">${esc(a.badgeTxt)}</span>
     </div>`
  ).join('') || '<div class="small">لا يوجد نشاط بعد</div>';

  const gaps = computeGaps().slice(0,4);
  document.getElementById('dashGaps').innerHTML = gaps.map(g=>
    `<div class="flex-between" style="border-bottom:1px solid var(--line); padding-bottom:8px;">
       <div style="font-size:12.5px;">${esc(g.category)}</div>
       <span class="badge b-review">${g.count} إشارة</span>
     </div>`
  ).join('') || '<div class="small">لا توجد فجوات مرصودة حاليًا</div>';
}

function computeGaps(){
  const byCat = {};
  state.failed.forEach(f=>{
    const c = f.category || 'غير مصنف';
    byCat[c] = (byCat[c]||0) + (f.count||1);
  });
  return Object.entries(byCat).map(([category,count])=>({category,count})).sort((a,b)=>b.count-a.count);
}

function renderKB(){
  const q = (document.getElementById('kbSearch').value||'').toLowerCase();
  const st = document.getElementById('kbStatusFilter').value;
  const ty = document.getElementById('kbTypeFilter').value;
  let rows = state.knowledge.filter(k=>{
    if(st && k.status!==st) return false;
    if(ty && k.contentType!==ty) return false;
    if(q){
      const hay = [k.title,k.question,(k.keywords||[]).join(' ')].join(' ').toLowerCase();
      if(!hay.includes(q)) return false;
    }
    return true;
  }).sort((a,b)=> (b.updatedAt||'').localeCompare(a.updatedAt||''));

  const body = document.getElementById('kbTableBody');
  document.getElementById('kbEmpty').style.display = rows.length? 'none':'block';
  body.innerHTML = rows.map(k=>`
    <tr class="row-click" onclick="openEditor('${k.id}')">
      <td><b>${esc(k.title)}</b><div class="small">${esc(k.service||'')}</div></td>
      <td>${esc(k.contentType||'—')}</td>
      <td>${esc(k.service||'—')}</td>

}

function kwSet(k){
  const text = [k.title, k.question, (k.keywords||[]).join(' ')].join(' ').toLowerCase();
  return new Set(text.split(/[\s،,.؛:\-]+/).filter(w=>w.length>2));
}
function jaccard(a,b){
  let inter=0; a.forEach(w=>{ if(b.has(w)) inter++; });
  const uni = new Set([...a,...b]).size;
  return uni? inter/uni : 0;
}
function findDuplicates(){
  const live = state.knowledge.filter(k=> ['Active','Approved','UnderReview'].includes(k.status));
  const pairs = [];
  for(let i=0;i<live.length;i++){
    for(let j=i+1;j<live.length;j++){
      const a=live[i], b=live[j];
      if(a.lineageId && a.lineageId===b.lineageId) continue;
      const sim = jaccard(kwSet(a), kwSet(b));
      if(sim >= 0.34) pairs.push({a,b,sim});
    }
  }
  return pairs.sort((x,y)=>y.sim-x.sim);
}
function findOutdated(){
  return state.knowledge.filter(k=>{
    if(k.status!=='Active') return false;
    if(k.expiryDate && k.expiryDate < todayStr()) return true;
    const days = daysSince(k.updatedAt||k.createdAt);
    return days!==null && days > 180;
  });
}

function lastPackage(){
  if(!state.packages.length) return null;
  return [...state.packages].sort((a,b)=> (b.issuedAt||'').localeCompare(a.issuedAt||''))[0];
}
function computePackageDiff(){
  const last = lastPackage();
  const cutoff = last ? last.issuedAt : null;
  const eligible = state.knowledge.filter(k=>
    (k.status==='Active' || k.status==='Approved') &&
    (!cutoff || (k.updatedAt||k.approvedAt||k.createdAt) > cutoff)
  );
  const added = eligible.filter(k=> (k.version||1)===1);
  const updated = eligible.filter(k=> (k.version||1)>1);
  const retracted = state.knowledge.filter(k=> k.status==='Retracted' && (!cutoff || (k.updatedAt||'') > cutoff));
  return {added, updated, retracted, cutoff};
}

async function issuePackage(){
  if(currentRole!=='specialist'){ toast('إصدار الحزمة يتطلب وضع مختص معتمِد', true); return; }
  const {added, updated, retracted} = computePackageDiff();
  if(added.length+updated.length+retracted.length===0){ toast('لا توجد تغييرات لإصدارها الآن', true); return; }

  const snapshot = [...added, ...updated].map(k=>({
    id:k.id, lineageId:k.lineageId||k.id, version:k.version||1, title:k.title, question:k.question,
    category:k.category, subCategory:k.subCategory, service:k.service, contentType:k.contentType,
    conditions:k.conditions||[], exceptions:k.exceptions||[], steps:k.steps||[],
    answerCustomer:k.answerCustomer, legalArticle:k.legalArticle||'', effectiveDate:k.effectiveDate||'',
    keywords:k.keywords||[], relatedQuestions:k.relatedQuestions||[]
  }));
  const retractedSnapshot = retracted.map(k=>({id:k.id, lineageId:k.lineageId||k.id, title:k.title}));

  const pkgDoc = {
    issuedAt: nowIso(), issuedBy: getUserName(),
    addedIds: added.map(k=>k.id), updatedIds: updated.map(k=>k.id), retractedIds: retracted.map(k=>k.id),
  };
  const ref = await db.collection('packages').add(pkgDoc);

  const fileObj = {
    packageId: ref.id, issuedAt: pkgDoc.issuedAt, issuedBy: pkgDoc.issuedBy,
    added: snapshot.filter(s=> added.some(a=>a.id===s.id)),
    updated: snapshot.filter(s=> updated.some(a=>a.id===s.id)),
    retracted: retractedSnapshot
  };
  const jsonStr = JSON.stringify(fileObj, null, 2);

  if(downloadsFn){
    try{
      await downloadsFn.save({filename:'ai-knowledge-package-'+todayStr()+'.json', data:jsonStr});
      toast('تم إصدار الحزمة وتنزيلها');
    }catch(e){
      toast('تم إصدار الحزمة، لكن تعذّر التنزيل ('+(e.code||'')+')', true);
    }
  }else{
    toast('تم إصدار الحزمة (التنزيل غير متاح في هذا العرض)');
  }
}

async function saveKnowledge(targetStatus){
  if(!dbAvailable){ toast('قاعدة البيانات غير متاحة في هذا العرض', true); return; }
  const vals = readFormValues();
  if(!vals.title){ toast('العنوان مطلوب', true); return; }
  const now = nowIso();
  try{
    if(editingId){
      const existing = state.knowledge.find(k=>k.id===editingId);
      await db.collection('knowledge').doc(editingId).update({
        ...vals, status: targetStatus || existing.status, updatedAt: now,
        reviewedBy: targetStatus==='UnderReview' ? getUserName() : (existing.reviewedBy||'')
      });
      toast('تم الحفظ');
    } else {
      const payload = {
        ...vals, status: targetStatus||'Draft', version:1, previousVersionId:null,
        createdBy:getUserName(), createdAt:now, updatedAt:now,
        approvedBy:'', approvedAt:'', reviewedBy: targetStatus==='UnderReview'?getUserName():''
      };
      const ref = await db.collection('knowledge').add(payload);
      await ref.update({lineageId: ref.id});
      if(pendingLink){
        if(pendingLink.type==='failed'){
          await db.collection('failedQuestions').doc(pendingLink.id).update({status:'تم التحويل لمعرفة', linkedKnowledgeId: ref.id});
        } else if(pendingLink.type==='correction'){
          await db.collection('corrections').doc(pendingLink.id).update({status:'تم التحويل', linkedKnowledgeId: ref.id});
        }
      }
      toast('تم إنشاء المعرفة');
    }
    closeEditor();
  }catch(e){
    toast('تعذّر الحفظ: '+(e.message||e.code||''), true);
  }
}

async function createNewVersion(oldId){
  const old = state.knowledge.find(k=>k.id===oldId);
  if(!old) return;
  const rest = {...old};
  delete rest.id; delete rest.status; delete rest.approvedBy; delete rest.approvedAt;
  delete rest.createdAt; delete rest.updatedAt; delete rest.version; delete rest.previousVersionId;
  const payload = {
    ...rest, status:'Draft', version:(old.version||1)+1, previousVersionId: old.id,
    lineageId: old.lineageId || old.id, createdBy:getUserName(), createdAt:nowIso(), updatedAt:nowIso(),
    approvedBy:'', approvedAt:'', reviewedBy:''
  };
  await db.collection('knowledge').add(payload);
  toast('تم إنشاء مسودة إصدار جديد (v'+payload.version+') في قاعدة المعرفة');
  closeEditor();
  go('kb');
}
async function retractKnowledge(id){
